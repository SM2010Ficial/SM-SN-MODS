#alias 3: TYPE_INTRO

sub ObjectMain

	switch Object.State
	case 0 //Load the Video
		ClearScreen(0)
		Object.State=1
		Object.Value0=512
		SetFade(0,0,0,255,0,255)
		break
		
	case 1 //Fade from Black
		if Object.Value0>0
			Object.Value0-=2
		else
			Object.State=2
			Object.Value1=0
		end if
		if Object.Value0==432
			PlayMusic(0)
			LoadVideo("Videos/Intro.rsv")
			NextVideoFrame()
		end if
		SetFade(0,0,0,Object.Value0,0,255)
		break
	
	case 2 //Play Video
		Object.Value0++
		if Object.Value0==4
			Object.Value1++
			Object.Value0=0
			NextVideoFrame()
		end if
		if Object.Value1==1008
			Object.State=3
		end if
		if KeyPress.Start==true
			Object.State=3
			Object.Value0=0
		end if
		break

	case 3 //Fade to White
		if Object.Value1<1008
			Music.Volume-=2
		end if
		Object.Value0+=4
		Object.Scale+=16
		if Object.Value0==320
			StopMusic()
			Stage.ListPos=1
			LoadStage()
		end if
		SetFade(255,255,255,Object.Value0,0,255)
		break
		
	end switch
	
end sub

sub ObjectDraw
	if Object.State<3
		DrawSpriteScreenXY(0, 212, 120)
	else
		DrawSpriteScreenFX(0, FX_SCALE, 212, 120)
	end if
end sub

sub ObjectStartup

	//Setup any Intro
	//Objects that are present in the level
	
	ArrayPos0=32
	while ArrayPos0<1056
		if Object[ArrayPos0].Type==TYPE_INTRO
			Object[ArrayPos0].Priority=1
			Object[ArrayPos0].iXPos=212
			Object[ArrayPos0].iYPos=120
		end if
		ArrayPos0++
	loop

	SpriteFrame(-128,-80,256,160,0,0)

	SetMusicTrack("Nexus/Intro.ogg",0,false)
end sub
